```html
    <link
      rel="stylesheet"

    />
    <!-- Importa os ícones do Google Fonts -->

      }

</head>

<body> <!-- Corpo da página -->

    <script>

      // Função responsável por abrir e fechar o menu

      function clickMenu() {

        // Obtém o elemento que possui id="itens"
        let itens = document.getElementById("itens");

        // Verifica se o menu está oculto
        if (getComputedStyle(itens).display == "none") {

          // Mostra o menu
          itens.style.display = "block";

        } else {

          // Esconde o menu
          itens.style.display = "none";

        }

      }

    </script>

</body>

</html>
```

```css
@charset "UTF-8";
/* Define a codificação de caracteres do arquivo CSS */

/* Reset básico */
* {
  font-family: Arial, Helvetica, sans-serif; /* Fonte padrão */
  margin: 0px; /* Remove margens */
  padding: 0px; /* Remove espaçamentos internos */
}

/* Configuração do fundo da página */
body,
html {
  background-color: rgb(110, 110, 107);
}

/* Cabeçalho */
header {
  background-color: gray;
}

/* Título principal */
header > h1 {
  padding: 10px;
  text-align: center;
}

/* Botão hamburguer */
#burguer {
  background-color: rgb(79, 75, 75);
  color: white;
  display: block;
  text-align: center;
  padding: 10px;
  cursor: pointer; /* Mãozinha ao passar o mouse */
}

/* Efeito ao passar o mouse */
#burguer:hover {
  background-color: rgb(183, 176, 176);
  color: black;
}

/* Menu oculto inicialmente */
menu {
  display: none;
}

/* Remove os marcadores da lista */
menu > ul {
  list-style-type: none;
}

/* Links do menu */
menu > ul > li > a {
  display: block;
  padding: 10px;
  text-decoration: none; /* Remove sublinhado */
  background-color: rgb(123, 123, 122);
  color: white;
  text-align: center;
  border-top: 2px solid rgb(103, 101, 101);
}

/* Efeito hover dos links */
menu > ul > li > a:hover {
  background-color: rgb(95, 94, 94);
  color: black;
}

/* Área principal do conteúdo */
main {
  width: 90vw; /* 90% da largura da tela */
  background-color: white;
  margin: auto; /* Centraliza */
  margin-top: 10px;
  padding: 10px;
  border-radius: 10px; /* Bordas arredondadas */
}

/* Título do artigo */
article > h2 {
  padding-bottom: 20px;
}

/* Parágrafos */
article > p {
  text-align: justify; /* Justifica o texto */
  margin-bottom: 20px;
  text-indent: 20px; /* Recuo da primeira linha */
}
```
